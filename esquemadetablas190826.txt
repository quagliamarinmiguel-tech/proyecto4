CREATE TABLE public.student (
  id bigint GENERATED ALWAYS AS IDENTITY NOT NULL,
  created_at timestamp with time zone NOT NULL DEFAULT now(),

  name text,
  last_name text,
  birthday date,
  phone bigint,
  email text,
  address text,
  birthplace text,
  education_level text,
  work boolean NOT NULL,
  current_work text NOT NULL,
  whatsapp_phone bigint NOT NULL,
  state smallint NOT NULL,
  img_id smallint,
  profile_id bigint,
  CONSTRAINT student_pkey PRIMARY KEY (id),
  CONSTRAINT student_profile_id_fkey FOREIGN KEY (profile_id) REFERENCES public.profile(id)
);
